3. Model Definition (MLP with pluggable activation)
We'll use a slightly larger MLP than for Wine, because Adult has more features and complexity.

python
import torch.nn as nn

class MLP(nn.Module):
    def __init__(self, input_dim, hidden_dims=[128, 64, 32], output_dim=2, activation_fn=None):
        super().__init__()
        layers = []
        prev_dim = input_dim
        for h_dim in hidden_dims:
            layers.append(nn.Linear(prev_dim, h_dim))
            if activation_fn is not None:
                layers.append(activation_fn())  # expects a callable returning a module
            prev_dim = h_dim
        layers.append(nn.Linear(prev_dim, output_dim))
        self.net = nn.Sequential(*layers)

    def forward(self, x):
        return self.net(x)
4. Wrapper for FTA/EFTA (MLP adaptation)
As before, we assume your FTA/EFTA classes are designed for 2D inputs (batch, channels, height, width). We'll adapt them for 1D features by unsqueezing to (batch, features, 1, 1).

python
class FTA_MLP(nn.Module):
    def __init__(self, depth, branch, in_features):
        super().__init__()
        self.fta = FTA(depth, branch, in_features)  # your existing class

    def forward(self, x):
        # x shape: (batch, in_features)
        x = x.unsqueeze(-1).unsqueeze(-1)  # (batch, in_features, 1, 1)
        x = self.fta(x)
        return x.squeeze(-1).squeeze(-1)   # back to (batch, in_features)
Similarly for EFTA. If your FTA class expects input of shape (batch, in_features) directly (e.g., you implemented a 1D version), you can skip the unsqueeze.

5. Training Function (same as before, but with possible small adjustments)
python
def train_model(model, train_loader, val_loader, epochs=100, lr=0.001, patience=10):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model.to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode='min', factor=0.5, patience=5)

    best_val_acc = 0.0
    best_epoch = 0
    counter = 0

    for epoch in range(epochs):
        # Training
        model.train()
        train_loss = 0.0
        for Xb, yb in train_loader:
            Xb, yb = Xb.to(device), yb.to(device)
            optimizer.zero_grad()
            out = model(Xb)
            loss = criterion(out, yb)
            loss.backward()
            optimizer.step()
            train_loss += loss.item()

        # Validation
        model.eval()
        val_correct = 0
        val_total = 0
        with torch.no_grad():
            for Xb, yb in val_loader:
                Xb, yb = Xb.to(device), yb.to(device)
                out = model(Xb)
                pred = out.argmax(dim=1)
                val_correct += (pred == yb).sum().item()
                val_total += yb.size(0)
        val_acc = val_correct / val_total * 100

        scheduler.step(100.0 - val_acc)  # ReduceLROnPlateau on validation error

        if val_acc > best_val_acc:
            best_val_acc = val_acc
            best_epoch = epoch
            counter = 0
            torch.save(model.state_dict(), 'best_model.pt')
        else:
            counter += 1
            if counter >= patience:
                print(f"Early stopping at epoch {epoch}")
                break

    # Load best and evaluate on test
    model.load_state_dict(torch.load('best_model.pt'))
    model.eval()
    test_correct = 0
    test_total = 0
    with torch.no_grad():
        for Xb, yb in test_loader:
            Xb, yb = Xb.to(device), yb.to(device)
            out = model(Xb)
            pred = out.argmax(dim=1)
            test_correct += (pred == yb).sum().item()
            test_total += yb.size(0)
    test_acc = test_correct / test_total * 100
    return test_acc
6. Run Multiple Seeds
We'll test the same set of activations as before, but you might want to include Swish, GELU, EELU as strong baselines.

python
import numpy as np
import random

def set_seed(seed):
    torch.manual_seed(seed)
    np.random.seed(seed)
    random.seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)

activations = {
    'ReLU': lambda: nn.ReLU(),
    'LeakyReLU': lambda: nn.LeakyReLU(0.01),
    'GELU': lambda: nn.GELU(),
    'Swish': lambda: nn.SiLU(),  # PyTorch's SiLU is Swish
    'Mish': lambda: nn.Mish(),
    'EELU': lambda: nn.ELU(alpha=1.0),  # EELU not standard; maybe use a custom, but ELU is close
    'Maxout (k=4)': lambda: MaxoutMLP(in_features=128, out_features=128, piece=4),  # need to define MaxoutMLP for hidden layers
    'FTA (d=2,k=2)': lambda: FTA_MLP(depth=2, branch=2, in_features=128),
    'FTA (d=3,k=2)': lambda: FTA_MLP(depth=3, branch=2, in_features=128),
    'FTA (d=2,k=3)': lambda: FTA_MLP(depth=2, branch=3, in_features=128),
    'EFTA (d=2,k=2)': lambda: EFTA_MLP(depth=2, branch=2, in_features=128),
    'EFTA (d=3,k=2)': lambda: EFTA_MLP(depth=3, branch=2, in_features=128),
    'EFTA (d=2,k=3)': lambda: EFTA_MLP(depth=2, branch=3, in_features=128),
}

input_dim = X_train_processed.shape[1]  # after preprocessing, ~108
n_seeds = 5   # 5–10 seeds; start with 5 for quick check
results = {}

for name, act_builder in activations.items():
    accs = []
    for seed in range(n_seeds):
        set_seed(seed)
        model = MLP(input_dim=input_dim, hidden_dims=[128, 64, 32], output_dim=2, activation_fn=act_builder)
        test_acc = train_model(model, train_loader, val_loader, epochs=100, patience=10)
        accs.append(test_acc)
        print(f"{name} seed {seed}: {test_acc:.2f}%")
    results[name] = accs

# Summary
for name, accs in results.items():
    mean = np.mean(accs)
    std = np.std(accs)
    print(f"{name}: {mean:.2f}% ± {std:.2f}%")

8. Additional Considerations
8.1 Maxout for MLP
If you want to include Maxout, define a module that applies Maxout to each hidden layer (since Maxout changes dimension, you need to integrate it carefully). A simple approach:

python
class MaxoutMLP(nn.Module):
    def __init__(self, in_features, out_features, piece=4):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.piece = piece
        self.linear = nn.Linear(in_features, out_features * piece)

    def forward(self, x):
        x = self.linear(x)
        x = x.view(-1, self.out_features, self.piece)
        x = x.max(dim=2)[0]
        return x
But then your MLP definition must be adjusted because Maxout replaces both linear and activation. For simplicity, you might skip Maxout or only compare ReLU, FTA, EFTA.

8.2 EELU Implementation
EELU is not in PyTorch core; you can approximate with ELU or implement a custom version. For a quick baseline, ELU is reasonable.

8.3 Statistical Testing
After collecting results, run a paired t-test (across seeds) to see if differences between top models are significant.
